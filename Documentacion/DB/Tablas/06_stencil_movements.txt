CREATE TABLE public.stencil_movements
(
    id_movement UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    id_stencil UUID NOT NULL,

    movement_type VARCHAR(20) NOT NULL,

    id_line SMALLINT,

    origin_location_id SMALLINT NOT NULL,

    destination_location_id SMALLINT NOT NULL,

    requested_by UUID NOT NULL,

    handled_by UUID,

    comments TEXT,

    movement_date TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    CONSTRAINT fk_stencil_movements_stencil
        FOREIGN KEY (id_stencil)
        REFERENCES public.stencils (id_stencil),

    CONSTRAINT fk_stencil_movements_line
        FOREIGN KEY (id_line)
        REFERENCES public.production_lines (id_line),

    CONSTRAINT fk_stencil_movements_origin
        FOREIGN KEY (origin_location_id)
        REFERENCES public.locations (id_location),

    CONSTRAINT fk_stencil_movements_destination
        FOREIGN KEY (destination_location_id)
        REFERENCES public.locations (id_location),

    CONSTRAINT fk_stencil_movements_requested_by
        FOREIGN KEY (requested_by)
        REFERENCES public.users (id_user),

    CONSTRAINT fk_stencil_movements_handled_by
        FOREIGN KEY (handled_by)
        REFERENCES public.users (id_user),

    CONSTRAINT ck_stencil_movements_type
        CHECK (movement_type IN ('OUT', 'IN', 'TRANSFER')),

    CONSTRAINT ck_stencil_movements_locations
        CHECK (origin_location_id <> destination_location_id)
);
