CREATE TABLE public.tray_inventory
(
    id_tray VARCHAR(30) NOT NULL,

    id_location SMALLINT NOT NULL,

    quantity INTEGER NOT NULL DEFAULT 0,

    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    CONSTRAINT pk_tray_inventory
        PRIMARY KEY (id_tray, id_location),

    CONSTRAINT fk_tray_inventory_tray
        FOREIGN KEY (id_tray)
        REFERENCES public.trays (id_tray),

    CONSTRAINT fk_tray_inventory_location
        FOREIGN KEY (id_location)
        REFERENCES public.locations (id_location),

    CONSTRAINT ck_tray_inventory_quantity
        CHECK (quantity >= 0)
);

CREATE INDEX idx_tray_inventory_location
ON public.tray_inventory (id_location);

CREATE INDEX idx_tray_inventory_quantity
ON public.tray_inventory (quantity);
