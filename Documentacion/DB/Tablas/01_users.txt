CREATE TABLE auth.users
(
    id_user UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    employee_number VARCHAR(20) NOT NULL,

    name VARCHAR(150) NOT NULL,

    password_hash TEXT NOT NULL,

    id_role SMALLINT NOT NULL,

    active BOOLEAN NOT NULL DEFAULT TRUE,

    last_login_at TIMESTAMPTZ,

    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    CONSTRAINT uq_users_employee_number
        UNIQUE (employee_number),

    CONSTRAINT fk_users_role
        FOREIGN KEY (id_role)
        REFERENCES auth.roles (id_role),

    CONSTRAINT ck_users_employee_number_not_empty
        CHECK (BTRIM(employee_number) <> ''),

    CONSTRAINT ck_users_name_not_empty
        CHECK (BTRIM(name) <> ''),

    CONSTRAINT ck_users_password_hash_not_empty
        CHECK (BTRIM(password_hash) <> '')
