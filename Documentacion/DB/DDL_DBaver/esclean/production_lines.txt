-- TABLA: LÍNEAS DE PRODUCCIÓN
-- ============================================================

CREATE TABLE esclean.production_lines
(
    id_line SMALLSERIAL PRIMARY KEY,

    code VARCHAR(20) NOT NULL,

    name VARCHAR(100) NOT NULL,

    active BOOLEAN NOT NULL DEFAULT TRUE,

    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    -- Código único de la línea
    CONSTRAINT uq_production_lines_code
        UNIQUE (code),

    -- No permitir código vacío
    CONSTRAINT ck_production_lines_code_not_empty
        CHECK (BTRIM(code) <> ''),

    -- No permitir nombre vacío
    CONSTRAINT ck_production_lines_name_not_empty
        CHECK (BTRIM(name) <> '')
);
