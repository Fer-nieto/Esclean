-- ============================================================
-- ESCLEAN
-- TABLA: TRAY INVENTORY
-- ============================================================
--
-- PROPÓSITO
-- ---------
-- Mantiene la existencia actual de charolas por ubicación.
--
-- UBICACIONES INICIALES:
--   CLEANING
--   CUARTERIAS
--   PISO
--   SCRAP
--
-- Cada charola puede tener una cantidad diferente en cada
-- ubicación.
--
-- ============================================================

CREATE TABLE esclean.tray_inventory
(
    id_tray VARCHAR(30) NOT NULL,

    location VARCHAR(30) NOT NULL,

    quantity INTEGER NOT NULL DEFAULT 0,

    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    CONSTRAINT pk_tray_inventory
        PRIMARY KEY (id_tray, location),

    CONSTRAINT fk_tray_inventory_tray
        FOREIGN KEY (id_tray)
        REFERENCES esclean.trays(id_tray),

    CONSTRAINT ck_tray_inventory_location
        CHECK (
            location IN (
                'CLEANING',
                'CUARTERIAS',
                'PISO',
                'SCRAP'
            )
        ),

    CONSTRAINT ck_tray_inventory_quantity
        CHECK (quantity >= 0)
);


-- ============================================================
-- COMENTARIOS
-- ============================================================

COMMENT ON TABLE esclean.tray_inventory IS
'Existencia actual de charolas por ubicación.';


COMMENT ON COLUMN esclean.tray_inventory.id_tray IS
'Identificador de la charola.';


COMMENT ON COLUMN esclean.tray_inventory.location IS
'Ubicación actual de la existencia de charolas.';


COMMENT ON COLUMN esclean.tray_inventory.quantity IS
'Cantidad actual de charolas en la ubicación.';


COMMENT ON COLUMN esclean.tray_inventory.updated_at IS
'Fecha y hora de la última actualización del inventario.';
