SELECT
    u.employee_number,
    u.name,
    r.name AS role,
    u.active,
    u.last_login_at,
    u.created_at,
    u.updated_at
FROM auth.users u
INNER JOIN auth.roles r
    ON r.id_role = u.id_role
ORDER BY
    r.name,
    u.name;
