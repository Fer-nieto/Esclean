CREATE OR REPLACE VIEW public.vw_stencil_inventory AS
SELECT
    s.id_stencil,
    s.steel_no,
    s.name,
    s.stencil_type,
    s.side,

    s.product_name,
    s.use_model,
    s.customer,

    s.manufacturer,
    s.manufacturing_no,
    s.order_by,

    s.store_location,
    s.replica,

    s.status,
    s.usable,

    s.id_location,
    l.code AS location_code,
    l.name AS location_name,

    s.registered_at,
    s.created_at,
    s.updated_at

FROM public.stencils AS s

INNER JOIN public.locations AS l
    ON l.id_location = s.id_location;
